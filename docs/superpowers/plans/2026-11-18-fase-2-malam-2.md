# Fase 2: Malam 2 — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Malam 2 ("Boneka di Bawah Lantai") bisa dimainkan dari akhir Malam 1 sampai kartu Malam 3. Isinya: tutorial kompas di kamar Ibu, papan kopong, kunci gudang dari Laras, Sosok Santet terlihat lewat celah gudang, linggis, boneka (M1, otomatis difoto), puzzle aksara Jawa, weton Ibu dan hitung mundur ke Selasa Kliwon, kunjungan Mbah Wiryo beserta pilihannya, lalu sekuens bersembunyi pertama (lemari, kolong ranjang, karung, menahan napas) dengan AI Sosok Santet. Catatan "Fase 2" di `CLAUDE.md` ikut diselesaikan.

**Architecture:** Aturan baru masuk Core dengan TDD: barang `KunciGudang`/`Linggis`, pilihan dan alur bacaan aksara (`AksaraJawa.Pilihan`, `BacaanAksara`), hitung mundur (`KalenderJawa`), otak Sosok Santet (`PerilakuSantet`, memakai `KewaspadaanSantet` yang sudah ada), dan waktu kedatangan Mbah Wiryo (`JamGim`). Runtime tetap tipis: komponen kecil yang dipakai lagi di malam berikutnya (`TempatSembunyi`, `SosokSantet` + `LantaiNavMesh`, `AmbilBarang`, `PapanLantai`, `PanelAksara`, `BonekaDiKotak`, `WajahIbu`) dan satu sutradara `AlurMalam2` yang hanya mengurutkan kejadian dan memanggil Core. Isi Malam 2 dipasang ke `Scene_RumahKeluarga` oleh skrip editor baru `PembuatMalam2` (denah K7 tidak berubah). NavMesh dibangun saat runtime dengan `UnityEngine.AI.NavMeshBuilder`, jadi batchmode tidak perlu menyimpan aset NavMesh dan ProjectSettings tidak berubah. Celah gudang adalah Timeline buatan skrip (`PembuatTimeline`), sisanya coroutine. Percakapan teks saja.

**Tech Stack:** Unity 6000.6.4f1, URP 17.6, Input System, uGUI 2.6 (legacy `Text`, `RawImage`), C# .NET Standard 2.1, Unity Test Framework 1.8.0, Timeline 6.6.0, modul AI bawaan (`NavMeshAgent`, `NavMeshBuilder`), Google Chrome headless (hanya untuk merender gambar aksara, Task 10).

```csharp
    public static class AksaraJawa
    {
        public static readonly string[] Ratmi = { "RA", "T", "MI" };
        public static readonly string[] SelasaKliwon = { "SE", "LA", "SA", "KLI", "WO", "N" };

        public static PuzzleUrutan<string> Buat(string[] kata) => new PuzzleUrutan<string>(kata, false);

        // Empat bacaan untuk tombol 1-4: jawaban dan tiga pengecoh yang bentuk atau bunyinya mirip. Urutan tetap.
        static readonly (string suku, string[] pilihan)[] DaftarPilihan =
        {
            ("RA", new[] { "KA", "RA", "HA", "NA" }),
            ("T", new[] { "N", "K", "T", "D" }),
            ("MI", new[] { "MA", "ME", "NI", "MI" }),
            ("SE", new[] { "SE", "SA", "SI", "LE" }),
            ("LA", new[] { "NA", "LA", "WA", "TA" }),
            ("SA", new[] { "PA", "YA", "SA", "SE" }),
            ("KLI", new[] { "KI", "LI", "KRI", "KLI" }),
            ("WO", new[] { "WA", "WO", "WE", "PO" }),
            ("N", new[] { "NG", "M", "T", "N" }),
        };

        public static string[] Pilihan(string suku)
        {
            foreach (var (s, pilihan) in DaftarPilihan)
```

```csharp
namespace TheSending.Core
{
    // Membaca tulisan di boneka (Game Story Malam 2): RATMI lalu SELASA KLIWON, suku demi suku.
    // Salah pilih tidak mengulang dan tidak ada penalti.
    public class BacaanAksara
    {
        public static readonly string[][] Kata = { AksaraJawa.Ratmi, AksaraJawa.SelasaKliwon };
        public static readonly int JumlahSuku = AksaraJawa.Ratmi.Length + AksaraJawa.SelasaKliwon.Length;

        PuzzleUrutan<string> puzzle = AksaraJawa.Buat(AksaraJawa.Ratmi);

        public int KataKe { get; private set; }
        public bool Selesai => KataKe >= Kata.Length;
        // Jumlah suku yang sudah terbaca dari awal; juga indeks gambar suku yang sedang ditandai.
        public int Terbaca { get; private set; }
        public string SukuSekarang => Selesai ? null : Kata[KataKe][puzzle.Kemajuan];

        public bool Pilih(string suku)
        {
            if (Selesai || !puzzle.Pilih(suku)) return false;
            Terbaca++;
            if (puzzle.Selesai && ++KataKe < Kata.Length) puzzle = AksaraJawa.Buat(Kata[KataKe]);
            return true;
        }
    }
}
```

```csharp
namespace TheSending.Core
{
    public enum KeadaanSantet { Patroli, Selidiki, Kejar }

    // Yang ditangkap indra Sosok Santet dalam satu frame. Jarak = jarak datar (meter) ke pemain.
    public struct Pengamatan
    {
        public float Jarak;
        public Gerak Gerak;
        public bool TahanNapas, MinyakSereh, Bersembunyi, SenterMenyala, SenterMengarah;
    }

    // Otak Sosok Santet (desain 2.4): Patroli, Selidiki (menuju suara atau cahaya terakhir), Kejar.
    // Penglihatannya buruk (Game Story 4.2): ia hanya mendengar langkah dan napas, dan melihat cahaya senter yang mengenainya.
    // Semua jarak dan lama adalah nilai awal; disetel saat playtest.
    public class PerilakuSantet
    {
        public const float JarakTangkap = 1f, JarakKejar = 3f, JarakLepas = 8f, LamaSelidiki = 6f;

        float sisaSelidiki;

        public KeadaanSantet Keadaan { get; private set; } = KeadaanSantet.Patroli;

        // Pemain yang bersembunyi hanya bisa ketahuan dari napasnya (Game Story 6.6).
        public static bool Mendeteksi(Pengamatan p) =>
            p.Bersembunyi
                ? KewaspadaanSantet.Terdengar(p.Jarak, Gerak.Diam, p.TahanNapas, p.MinyakSereh)
                : KewaspadaanSantet.Terdengar(p.Jarak, p.Gerak, p.TahanNapas, p.MinyakSereh)
                  || KewaspadaanSantet.CahayaTerlihat(p.Jarak, p.SenterMenyala, p.SenterMengarah);

        // true = pemain tertangkap pada frame ini.
        public bool Tick(float detik, Pengamatan p)
        {
            if (Mendeteksi(p))
            {
                if (p.Jarak <= JarakTangkap) return true;
                bool kejar = !p.Bersembunyi && (p.Jarak <= JarakKejar || Keadaan == KeadaanSantet.Kejar);
                Keadaan = kejar ? KeadaanSantet.Kejar : KeadaanSantet.Selidiki;
                sisaSelidiki = LamaSelidiki;
                return false;
            }
            if (Keadaan == KeadaanSantet.Kejar && (p.Bersembunyi || p.Jarak > JarakLepas))
            {
                Keadaan = KeadaanSantet.Selidiki;
                sisaSelidiki = LamaSelidiki;
            }
            else if (Keadaan == KeadaanSantet.Selidiki)
            {
                sisaSelidiki -= detik;
                if (sisaSelidiki <= 0f) Keadaan = KeadaanSantet.Patroli;
            }
            return false;
        }
    }
}
```

```csharp
namespace TheSending.Runtime
{
    // Lemari, kolong ranjang, atau balik tumpukan karung (Game Story 6.6). E masuk, E lagi keluar.
    // Jalan setelah PemainInteraksi (urutan 50): E yang mengeluarkan pemain tidak ikut memasukkannya lagi.
    [DefaultExecutionOrder(50)]
    public class TempatSembunyi : Interaktif
    {
        [SerializeField] PemainGerak pemain;
        [SerializeField] Transform titikDalam, titikKeluar;

        bool dipakai;
        int frameMasuk;

        public override bool BisaDiinteraksi =>
            pemain != null && titikDalam != null && titikKeluar != null && !pemain.Bersembunyi;

        public override void Interaksi()
        {
            pemain.Sembunyi(titikDalam);
            dipakai = true;
            frameMasuk = Time.frameCount;   // E yang dipakai masuk tidak ikut mengeluarkan
        }

        void Update()
        {
            if (!dipakai) return;
            if (!pemain.Bersembunyi)        // dipindah sutradara (adegan, muat ulang)
            {
                dipakai = false;
                return;
            }
            var gm = GameManager.Instance;
            if (gm == null || gm.Dijeda || Time.frameCount == frameMasuk || !gm.Kontrol.Interaksi.WasPressedThisFrame()) return;
            dipakai = false;
            pemain.Pindah(titikKeluar);
        }
    }
}
```

```csharp
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.AI;

namespace TheSending.Runtime
{
    // NavMesh lantai scene, dibangun saat Sosok Santet pertama kali muncul (runtime), supaya pembuat scene di batchmode
    // tidak perlu menyimpan aset NavMesh. Setelan agen sendiri (radius 0,25 m) supaya pintu rumah bisa dilewati tanpa
    // mengubah ProjectSettings. Hanya collider padat yang dihitung: trigger (zona) dan objek yang diabaikan tidak.
    public class LantaiNavMesh : MonoBehaviour
    {
        public const float RadiusAgen = 0.25f, TinggiAgen = 2f;

        [SerializeField] Vector3 ukuran = new Vector3(30f, 8f, 30f);

        NavMeshDataInstance instance;
        bool siap;

        public int AgentTypeID { get; private set; } = -1;

        // abaikan: pemain (CharacterController) dan Sosok sendiri, supaya tidak meninggalkan lubang di NavMesh.
        public void Siapkan(params Transform[] abaikan)
        {
            if (siap) return;
            var setelan = NavMesh.CreateSettings();
            setelan.agentRadius = RadiusAgen;
            setelan.agentHeight = TinggiAgen;
            setelan.agentClimb = 0.3f;
            setelan.agentSlope = 45f;
            AgentTypeID = setelan.agentTypeID;

            var batas = new Bounds(transform.position, ukuran);
            var sumber = new List<NavMeshBuildSource>();
            NavMeshBuilder.CollectSources(batas, ~0, NavMeshCollectGeometry.PhysicsColliders, 0, new List<NavMeshBuildMarkup>(), sumber);
            sumber.RemoveAll(x => x.component is Collider c && (c.isTrigger || Diabaikan(c.transform, abaikan)));
            var data = NavMeshBuilder.BuildNavMeshData(setelan, sumber, batas, Vector3.zero, Quaternion.identity);
            if (data == null)
            {
                Debug.LogError("NavMesh gagal dibangun.", this);
                return;
            }
            instance = NavMesh.AddNavMeshData(data);
            siap = true;
        }

        static bool Diabaikan(Transform t, Transform[] abaikan)
        {
            foreach (var a in abaikan)
                if (a != null && t.IsChildOf(a)) return true;
            return false;
        }

        void OnDestroy()
        {
            if (siap) instance.Remove();
            if (AgentTypeID != -1) NavMesh.RemoveSettings(AgentTypeID);
        }
    }
}
```

```csharp
using System;
using TheSending.Core;
using UnityEngine;
using UnityEngine.AI;

namespace TheSending.Runtime
{
    // Sosok Santet di scene: NavMeshAgent yang digerakkan PerilakuSantet (Core). Nyalakan objeknya untuk memulai patroli.
    // Kecepatan selidik dan kejar naik sesuai tingkat kekuatan (desain 4.6). Diam saat gim dijeda (Time.timeScale 0).
    // PembuatMalam2 memasang NavMeshAgent dalam keadaan mati; agen dinyalakan di sini setelah NavMesh siap.
    [RequireComponent(typeof(NavMeshAgent))]
    public class SosokSantet : MonoBehaviour
    {
        public const float KecepatanPatroli = 1.2f;
        const float JarakTiba = 0.4f, TinggiDada = 1.4f;
        const float SudutSenter = 25f;   // separuh spotAngle senter HP (50°, PembuatRig)

        [SerializeField] LantaiNavMesh lantai;
        [SerializeField] PemainGerak pemain;
        [SerializeField] SenterHP senter;
        [SerializeField] Transform[] titikPatroli = new Transform[0];
        [SerializeField] Suara gemerincing;   // gemerincing paku setiap kali bergerak (Game Story 4.2), loop 3D

        NavMeshAgent agen;
        PerilakuSantet otak = new PerilakuSantet();
        int titikKe;
        Vector3 sasaranSelidiki;

        public event Action Menangkap;
        public KeadaanSantet Keadaan => otak.Keadaan;

        void Awake() => agen = GetComponent<NavMeshAgent>();

        void OnEnable()
        {
            otak = new PerilakuSantet();
            titikKe = 0;
            if (lantai == null || pemain == null || titikPatroli.Length == 0)
            {
                Debug.LogError("SosokSantet butuh lantai, pemain, dan titik patroli.", this);
                enabled = false;
                return;
            }
            lantai.Siapkan(transform, pemain.transform);
            agen.agentTypeID = lantai.AgentTypeID;
            agen.enabled = true;
            if (!agen.Warp(transform.position))
            {
                Debug.LogError($"SosokSantet di {transform.position} tidak berada di NavMesh.", this);
                enabled = false;
                return;
            }
            agen.speed = KecepatanPatroli;
            agen.SetDestination(titikPatroli[0].position);
            if (gemerincing != null) gemerincing.Putar();
        }

        void OnDisable()
        {
            if (gemerincing != null) gemerincing.Hentikan();
            if (agen != null) agen.enabled = false;
        }

        void Update()
        {
            var gm = GameManager.Instance;
            if (gm == null || gm.Dijeda || !agen.isOnNavMesh) return;
            var s = gm.State;
            var p = Amati(s);
            var sebelumnya = otak.Keadaan;
            if (otak.Tick(Time.deltaTime, p))
            {
                enabled = false;
                Menangkap?.Invoke();
                return;
            }
            if (PerilakuSantet.Mendeteksi(p)) sasaranSelidiki = Datar(pemain.transform.position);
            float cepat = KewaspadaanSantet.KecepatanKejar(KewaspadaanSantet.TingkatKekuatan(s));
            switch (otak.Keadaan)
            {
                case KeadaanSantet.Kejar:
                    agen.speed = cepat;
                    agen.SetDestination(Datar(pemain.transform.position));
                    break;
                case KeadaanSantet.Selidiki:
                    agen.speed = cepat;
                    agen.SetDestination(sasaranSelidiki);
                    break;
                default:
                    agen.speed = KecepatanPatroli;
                    if (sebelumnya != KeadaanSantet.Patroli) agen.SetDestination(titikPatroli[titikKe].position);
                    else if (!agen.pathPending && agen.remainingDistance <= JarakTiba)
                    {
                        titikKe = (titikKe + 1) % titikPatroli.Length;
                        agen.SetDestination(titikPatroli[titikKe].position);
                    }
                    break;
            }
        }

        // Titik pemain di ketinggian Sosok: titik dalam kolong ranjang berada di bawah lantai.
        Vector3 Datar(Vector3 v) => new Vector3(v.x, transform.position.y, v.z);

        Pengamatan Amati(GameState s)
        {
            Vector3 a = transform.position, b = pemain.transform.position;
            bool menyala = senter != null && senter.Menyala;
            return new Pengamatan
            {
                Jarak = new Vector2(a.x - b.x, a.z - b.z).magnitude,
                Gerak = pemain.GerakSekarang,
                TahanNapas = pemain.Napas.SedangMenahan,
                MinyakSereh = s.PunyaMinyakSereh,
                Bersembunyi = pemain.Bersembunyi,
                SenterMenyala = menyala,
                SenterMengarah = menyala && Disorot(),
            };
        }

        // Cahaya senter mengenai Sosok: ia di dalam kerucut senter dan tidak terhalang dinding.
        bool Disorot()
        {
            var mata = pemain.Kamera.position;
            var arah = transform.position + Vector3.up * TinggiDada - mata;
            if (Vector3.Angle(pemain.Kamera.forward, arah) > SudutSenter) return false;
            return !Physics.Raycast(mata, arah.normalized, out var kena, arah.magnitude, ~0, QueryTriggerInteraction.Ignore)
                   || kena.transform.IsChildOf(transform);
        }
    }
}
```

```csharp
        [SerializeField] Renderer[] kulit = new Renderer[0];

        MaterialPropertyBlock blok;
        Color[] asli;
        int terakhir = -1;

        void Awake()
        {
            blok = new MaterialPropertyBlock();
            asli = Array.ConvertAll(kulit, r => r.sharedMaterial.GetColor(WarnaDasar));
        }

        void Update()
        {
            var gm = GameManager.Instance;
            if (gm == null || gm.State.KondisiIbu == terakhir) return;
            terakhir = gm.State.KondisiIbu;
            float pucat = 1f - Mathf.Clamp01(terakhir / (float)KondisiIbu.Awal);
            for (int i = 0; i < kulit.Length; i++)
            {
                blok.SetColor(WarnaDasar, Color.Lerp(asli[i], Pucat, pucat));
                kulit[i].SetPropertyBlock(blok);
            }
        }
    }
}
```

Kode Unicode (blok Javanese U+A980–U+A9DF): ha A9B2, na A9A4, ca A995, ra A9AB, ka A98F, da A9A2, ta A9A0, sa A9B1, wa A9AE, la A9AD, pa A9A5, dha A99D, ja A997, ya A9AA, nya A99A, ma A9A9, ga A992, ba A9A7, tha A99B, nga A994; wulu A9B6, taling A9BA, tarung A9B4, pangkon A9C0. Suku: RA = ꦫ, T = ꦠ꧀, MI = ꦩꦶ, SE = ꦱꦺ, LA = ꦭ, SA = ꦱ, KLI = ꦏ꧀ꦭꦶ (pasangan la + wulu), WO = ꦮꦺꦴ, N = ꦤ꧀.

- [ ] **Step 1: `tools/render-aksara.sh`**

```bash
#!/usr/bin/env bash
# Gambar aksara Jawa untuk puzzle Malam 2 (PanelAksara), dirender Google Chrome headless dengan font Noto Sans Javanese,
# supaya sandhangan dan pasangan tersusun benar (uGUI tidak bisa menyusun aksara Jawa).
# Hasil: Assets/_Project/Art/UI/Aksara/Suku_<n>.png (tulisan di boneka) dan Tabel.png (Halaman 2).
# Pemakaian: bash tools/render-aksara.sh   (jalankan ulang setelah mengubah isi di bawah)
set -euo pipefail
AKAR="$(cd "$(dirname "$0")/.." && pwd)"
FONT="$AKAR/Assets/_Project/Art/Fonts/Resources/NotoSansJavanese-Regular.ttf"
KELUAR="$AKAR/Assets/_Project/Art/UI/Aksara"
CHROME="/Applications/Google Chrome.app/Contents/MacOS/Google Chrome"
[ -f "$FONT" ] || { echo "Font belum ada: $FONT (lihat docs/aset/fase-2.md)"; exit 1; }
[ -x "$CHROME" ] || { echo "Google Chrome tidak ditemukan: $CHROME"; exit 1; }
mkdir -p "$KELUAR"
SEMENTARA="$(mktemp -d)"
trap 'rm -rf "$SEMENTARA"' EXIT
URL_FONT="file://${FONT// /%20}"

# render <nama> <lebar> <tinggi> <css tambahan> <isi html>
render() {
  cat > "$SEMENTARA/$1.html" <<EOF
<!DOCTYPE html><html><head><meta charset="utf-8"><style>
@font-face { font-family: Jawa; src: url("$URL_FONT"); }
html, body { margin: 0; background: transparent; color: #E6E0D1; }
body { width: ${2}px; height: ${3}px; font-family: Jawa, sans-serif; }
$4
</style></head><body>$5</body></html>
EOF
  rm -f "$KELUAR/$1.png"
  "$CHROME" --headless=new --disable-gpu --hide-scrollbars --allow-file-access-from-files \
    --default-background-color=00000000 --virtual-time-budget=3000 \
    --window-size="$2,$3" --screenshot="$KELUAR/$1.png" "file://$SEMENTARA/$1.html" >/dev/null 2>&1 || true
  [ -s "$KELUAR/$1.png" ] || { echo "Gagal membuat $1.png"; exit 1; }
  echo "  $1.png"
}

# Tulisan di boneka, urutan sama dengan BacaanAksara.Kata: RA T MI | SE LA SA KLI WO N
SUKU=( "&#xA9AB;" "&#xA9A0;&#xA9C0;" "&#xA9A9;&#xA9B6;"
       "&#xA9B1;&#xA9BA;" "&#xA9AD;" "&#xA9B1;" "&#xA98F;&#xA9C0;&#xA9AD;&#xA9B6;" "&#xA9AE;&#xA9BA;&#xA9B4;" "&#xA9A4;&#xA9C0;" )
for i in "${!SUKU[@]}"; do
  render "Suku_$i" 256 256 \
    "body { display: flex; align-items: center; justify-content: center; font-size: 130px; line-height: 1; }" \
    "<span>${SUKU[$i]}</span>"
done

# Tabel Halaman 2: 20 aksara dasar dan sandhangan yang dipakai tulisan di boneka (contoh dengan ka).
SEL=""
for p in "A9B2 ha" "A9A4 na" "A995 ca" "A9AB ra" "A98F ka" "A9A2 da" "A9A0 ta" "A9B1 sa" "A9AE wa" "A9AD la" \
         "A9A5 pa" "A99D dha" "A997 ja" "A9AA ya" "A99A nya" "A9A9 ma" "A992 ga" "A9A7 ba" "A99B tha" "A994 nga"; do
  set -- $p
  SEL="$SEL<div><b>&#x$1;</b>$2</div>"
done
SANDHANGAN="<div><b>&#xA98F;&#xA9B6;</b>ki (wulu)</div><div><b>&#xA98F;&#xA9BA;</b>ke (taling)</div><div><b>&#xA98F;&#xA9BA;&#xA9B4;</b>ko (taling tarung)</div><div><b>&#xA98F;&#xA9C0;</b>k (pangkon)</div><div><b>&#xA98F;&#xA9C0;&#xA9AD;</b>kla (pasangan la)</div>"
render "Tabel" 768 960 \
  "body { padding: 24px; box-sizing: border-box; } h1, h2 { font-family: sans-serif; font-size: 30px; margin: 0 0 10px; }
   .g { display: grid; grid-template-columns: repeat(5, 1fr); gap: 4px; margin-bottom: 18px; }
   .g div { text-align: center; font-family: sans-serif; font-size: 24px; }
   .g b { display: block; font-family: Jawa; font-size: 62px; font-weight: normal; line-height: 1.35; }" \
  "<h1>Aksara Jawa</h1><div class=\"g\">$SEL</div><h2>Sandhangan</h2><div class=\"g\">$SANDHANGAN</div>"
echo "Selesai: $KELUAR"
```

```csharp
using System;
using System.IO;
using TheSending.Core;
using TheSending.Runtime;
using UnityEditor;
using UnityEngine;

namespace TheSending.EditorTools
{
    // Aset dialog Malam 2, isi Jurnal yang terbuka di Malam 2, dan impor gambar aksara (tools/render-aksara.sh).
    // Tim menyunting teks DI FILE INI lalu menjalankan The Sending > Bangun Ulang Semua (K15).
    // Batchmode: -executeMethod TheSending.EditorTools.PembuatDataFase2.Buat
    public static class PembuatDataFase2
    {
        const string Folder = "Assets/_Project/Data/Fase2";
        const string JalurJurnal = "Assets/_Project/Data/Jurnal.asset";
        const string FolderAksara = "Assets/_Project/Art/UI/Aksara";
        const int UkuranTabel = 1024;   // tabel Halaman 2 harus terbaca; batas PSX 256 hanya untuk tekstur dunia

        internal const string LarasKunciGudang = "LarasKunciGudang", Weton = "Weton", WiryoDatang = "WiryoDatang",
            WiryoBisikan = "WiryoBisikan", KalenderMalam2 = "KalenderMalam2";

        [MenuItem("The Sending/Buat Data Fase 2")]
        public static void Buat()
        {
            Directory.CreateDirectory(Folder);
            AssetDatabase.Refresh();

            Simpan(LarasKunciGudang,
                B("Laras", "Mas, kamar Ibu bau anyir dari tadi sore. Aku nggak berani masuk."),
                B("Raka", "Ada papan lantai yang bunyinya kopong di dekat ranjang. Kita punya linggis?"),
                B("Laras", "Ada di gudang. Ini kuncinya, Mas.", efek: "barang:KunciGudang"));
            Simpan(Weton,
                B("Raka", "Ra... t... mi. Ratmi. Ini nama Ibu."),
                B("Raka", "Dan yang ini... Se-la-sa Kli-won."),
                B("Laras", "Selasa Kliwon... itu weton Ibu, Mas."),                       // dikutip
                B("Raka", "Malam ini Kamis Kliwon. Selasa Kliwon... lima malam lagi."));
            Simpan(WiryoDatang,
                B("Mbah Wiryo", "Kulonuwun... Mbah dengar ibumu sakit, Le.", "Permisi... Mbah dengar ibumu sakit, Nak."),
                B("Raka", "Mbah Wiryo? Malam-malam begini?"),
                P(B("Mbah Wiryo", "Barang begini jangan disimpan di rumah, Le. Biar Mbah yang larung ke kali, supaya ibumu tenang."),   // dikutip
                    ("Serahkan boneka kepada Mbah Wiryo", "serahkan:M1; tujuan:PilihanWiryo", 3),
                    ("Simpan boneka sendiri", "tujuan:PilihanWiryo", 5)),
                B("Raka", "Iya, Mbah. Tolong dibuang saja."),
                B("Mbah Wiryo", "Nah, begitu. Ibumu biar Mbah yang mendoakan.", akhir: true),
                B("Raka", "Tidak usah, Mbah. Biar saya simpan dulu."),
                B("Mbah Wiryo", "(tersenyum tipis) Yo wis, sakarepmu, Le.", "Ya sudah, terserah kamu, Nak."));
            Simpan(WiryoBisikan,
                B("Mbah Wiryo (kepada Laras)", "Ingat ya, Nduk.", "Ingat ya, Nak."));                   // dikutip
            Simpan(KalenderMalam2,
                B("Raka", "Rabu, Malam Kamis Kliwon."));

            IsiJurnal();
            SiapkanGambarAksara();
            AssetDatabase.SaveAssets();
            Debug.Log($"Data Fase 2 dibuat: 5 dialog di {Folder}, isi Jurnal Malam 2.");
        }

        internal static DataDialog Muat(string nama) =>
            AssetDatabase.LoadAssetAtPath<DataDialog>($"{Folder}/{nama}.asset")
            ?? throw new FileNotFoundException($"Dialog Fase 2 belum ada: {Folder}/{nama}.asset. Jalankan PembuatDataFase2.Buat.");

        // null jika gambar belum dibuat (tools/render-aksara.sh): PanelAksara memakai teks pengganti.
        internal static Texture2D Tabel() => AssetDatabase.LoadAssetAtPath<Texture2D>($"{FolderAksara}/Tabel.png");

        internal static Texture2D[] GambarSuku()
        {
            var hasil = new Texture2D[BacaanAksara.JumlahSuku];
            for (int i = 0; i < hasil.Length; i++)
            {
                hasil[i] = AssetDatabase.LoadAssetAtPath<Texture2D>($"{FolderAksara}/Suku_{i}.png");
                if (hasil[i] == null) return null;
            }
            return hasil;
        }

        static BarisDialog B(string pembicara, string teks, string terjemahan = "", string efek = "", bool akhir = false) =>
            new BarisDialog { Pembicara = pembicara, Teks = teks, Terjemahan = terjemahan, Efek = efek, Akhir = akhir };

        // Baris dengan pilihan: (teks, kode efek, indeks baris berikutnya).
        static BarisDialog P(BarisDialog baris, params (string teks, string efek, int lompatKe)[] pilihan)
        {
            baris.Pilihan = Array.ConvertAll(pilihan, p => new PilihanDialog { Teks = p.teks, Efek = p.efek, LompatKe = p.lompatKe });
            return baris;
        }

        // Baris terakhir selalu menutup percakapan. Aset yang sudah ada diisi ulang di tempat supaya GUID tetap.
        static void Simpan(string nama, params BarisDialog[] baris)
        {
            foreach (var b in baris)
                if (string.IsNullOrWhiteSpace(b.Teks)) throw new InvalidDataException($"Dialog '{nama}': ada baris tanpa teks.");
            baris[baris.Length - 1].Akhir = true;
            var masalah = PemeriksaDialog.Periksa(baris);
            if (masalah.Count > 0) throw new InvalidDataException($"Dialog '{nama}': {string.Join("; ", masalah)}");

            string jalur = $"{Folder}/{nama}.asset";
            var data = AssetDatabase.LoadAssetAtPath<DataDialog>(jalur);
            if (data == null)
            {
                data = ScriptableObject.CreateInstance<DataDialog>();
                AssetDatabase.CreateAsset(data, jalur);
            }
            data.Baris = baris;
            EditorUtility.SetDirty(data);
        }

        // Jurnal.asset dibuat PembuatSandbox (0B). Isi Media M1 sengaja tidak menyebut bacaan aksara (itu jawaban puzzle).
        static void IsiJurnal()
        {
            var j = AssetDatabase.LoadAssetAtPath<DataJurnal>(JalurJurnal)
                ?? throw new FileNotFoundException($"{JalurJurnal} belum ada. Jalankan PembuatSandbox.Buat sekali.");
            j.Media[Kode.M1].Isi =
                "Ditemukan di bawah papan lantai kamar Ibu. Kain putih kumal berlilit benang merah, jarum menusuk " +
                "kepala dan perutnya. Ada tulisan aksara Jawa di tubuhnya.";
            j.CatatanBapak[0].Isi =
                "Orang Jawa memakai dua hitungan hari sekaligus: tujuh hari (Senin sampai Minggu) dan lima hari pasaran " +
                "(Legi, Pahing, Pon, Wage, Kliwon). Gabungan hari dan pasaran saat seseorang lahir disebut weton, misalnya " +
                "Selasa Kliwon. Gabungan yang sama berulang setiap 35 hari. Weton dipakai dalam tradisi, misalnya untuk " +
                "menentukan hari baik hajatan.";
            j.CatatanBapak[1].Isi =
                "Aksara Jawa (hanacaraka) punya 20 aksara dasar. Setiap aksara berbunyi konsonan dengan vokal a: ha, na, " +
                "ca, ra, ka, dan seterusnya. Sandhangan mengubah bunyinya: wulu menjadi i, taling menjadi e, taling tarung " +
                "menjadi o. Pangkon mematikan vokal, dan pasangan dipakai jika konsonan mati diikuti konsonan lain.";
            j.CatatanBapak[6].Isi =
                "Larung berarti menghanyutkan sesaji ke laut atau sungai. Di banyak daerah di Jawa, larung adalah upacara " +
                "syukur dan doa keselamatan bersama. Tradisi ini dihormati masyarakat dan tidak ada hubungannya dengan santet.";
            EditorUtility.SetDirty(j);
        }

        static void SiapkanGambarAksara()
        {
            if (!(AssetImporter.GetAtPath($"{FolderAksara}/Tabel.png") is TextureImporter imp) || imp.maxTextureSize == UkuranTabel) return;
            imp.maxTextureSize = UkuranTabel;
            imp.SaveAndReimport();
        }
    }
}
```
